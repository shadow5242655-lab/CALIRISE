name: Unzip Website

on:
  workflow_dispatch:

permissions:
  contents: write

jobs:
  unzip:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout main branch
        uses: actions/checkout@v4
        with:
          ref: main
          fetch-depth: 0
          persist-credentials: true

      - name: Ensure project.zip exists
        run: |
          if [ ! -f project.zip ]; then
            echo "project.zip not found in repository root" >&2
            exit 1
          fi

      - name: Unzip files
        run: |
          unzip -o project.zip -d ./

      - name: Remove dangerous files
        run: |
          # Remove any .env files (anywhere in tree)
          find . -type f -name ".env*" -exec rm -f {} + || true
          # Remove any nested .git folders extracted from the archive, but keep the workflow runner's ./.git
          find . -type d -name ".git" -not -path "./.git" -prune -exec rm -rf {} + || true

      - name: Commit and push
        env:
          GIT_AUTHOR_NAME: "github-actions[bot]"
          GIT_AUTHOR_EMAIL: "github-actions[bot]@users.noreply.github.com"
        run: |
          git config user.name "$GIT_AUTHOR_NAME"
          git config user.email "$GIT_AUTHOR_EMAIL"
          git add -A
          # If there are no staged changes, exit cleanly
          if git diff --staged --quiet; then
            echo "No changes to commit"
            exit 0
          fi
          git commit -m "Auto-extract website from project.zip"
          git push origin HEAD:main
